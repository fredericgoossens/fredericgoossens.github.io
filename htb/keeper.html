Root access

  1. Try root credentials using ssh or su → not working
  2. Split putty .ppk into openssh keys
    # Install putty tools
sudo apt-get install putty-tools

puttygen oi.ppk -O private-openssh -o id_dsa      # private key
puttygen oi.ppk -O public-openssh -o id_dsa.pub.  # public key

# Permissions
chmod 600 id_dsa
chmod 666 id_dsa.pub
  3. ssh and specify key
ssh root@10.10.11.227 -i Desktop/id_dsa
